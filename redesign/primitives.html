02 / TypeScript сигнатури

Точни props на всеки компонент.

Това е договор. Когато Codex имплементира компонент, тези props са задължителни.

01 · Surface · primitives/Surface.tsx
/** * Surface — основна обвивка. Всяка card в продукта започва с Surface. * Не съдържа padding/typography — само фон, border, radius, shadow. */ import { ReactNode, HTMLAttributes } from 'react'; export interface SurfaceProps extends HTMLAttributes<HTMLDivElement> { /** Цветова семантика */ variant?: 'paper' | 'paper-deep' | 'scene' | 'scene-deep'; /** Размер на radius — 'card' (22px), 'tile' (14px), 'none' (0) */ radius?: 'card' | 'tile' | 'none'; /** Включи shadow */ elevation?: 'none' | 'card' | 'scene'; /** Render като различен HTML element */ as?: 'div' | 'section' | 'article' | 'aside'; children: ReactNode; } // Default: variant='paper', radius='tile', elevation='card', as='div'
02-03 · PaperCard & SceneCard
/** * PaperCard / SceneCard — Surface + standardized padding + slot за meta + body. * Заглавието е винаги слот, не prop, за да поддържа богат markup. */ export interface PaperCardProps { /** Eyebrow text (mono, all-caps) над заглавието */ eyebrow?: string; /** Размер на padding — 'sm' (16px), 'md' (28px), 'lg' (48px) */ density?: 'sm' | 'md' | 'lg'; /** Право-aligned meta info вдясно от заглавието */ meta?: ReactNode; children: ReactNode; } // SceneCard има идентичен API, само че variant='scene'. // Употреба: <PaperCard eyebrow="Досие" meta={<Eyebrow>14.05</Eyebrow>}> <Display>Селото оцеля</Display> <p>Върколак</p> </PaperCard>
04 · Eyebrow · primitives/Eyebrow.tsx
/** * Eyebrow — кратък all-caps mono label. * Винаги стои НАД heading-а, не вместо него. * Цвят се определя от тона на родителя (Surface). */ export interface EyebrowProps { /** Сила на тона — 'default' (gold), 'muted' (ink-faint), 'blood' */ tone?: 'default' | 'muted' | 'blood'; /** Render като span (inline) или div (block, with margin) */ inline?: boolean; children: ReactNode; } // Винаги Bulgarian, винаги ≤ 4 думи, винаги в Cyrillic capitalization.
05 · Display · primitives/Display.tsx
/** * Display — основното heading семейство. * Винаги Cormorant Garamond. Никога не bold над 500. */ export interface DisplayProps { /** Размерен tier — мапва се към --text-display/h1/h2/h3 */ size?: 'hero' | 'h1' | 'h2' | 'h3' | 'h4'; /** HTML семантика (отделена от visual size) */ as?: 'h1' | 'h2' | 'h3' | 'h4'; /** Активен italic accent на части от текста — wrap с <em> */ children: ReactNode; } // Default: size='h2', as=auto from size. // <em> вътре в Display получава gold курсив автоматично.
06 · Pill · primitives/Pill.tsx
/** * Pill — универсален бутон / линк / chip. * Един компонент, четири роли. Никога два компонента за „button-like". */ export interface PillProps { /** Визуална тежест */ variant?: 'primary' | 'ghost' | 'oauth-google' | 'oauth-discord'; /** Размер */ size?: 'sm' | 'md' | 'lg'; /** Ако подаваш href → render като <a>; иначе <button> */ href?: string; /** Loading state със spinner */ loading?: boolean; /** Префикс символ — никога emoji, само геометричен глиф */ leading?: 'play' | 'arrow' | 'check' | 'cross'; disabled?: boolean; onClick?: () => void; children: ReactNode; } // Disabled не само opacity — pointer-events: none + focus skip. // Primary винаги blood; ghost винаги bordered с тон от родителя.
07 · Medallion · primitives/Medallion.tsx
/** * Medallion — кръгъл златен елемент с цифра или глиф. * Сигнатурният елемент на бранда. Употреба пестелива. */ export interface MedallionProps { /** Размер — 'sm' (40px), 'md' (76px), 'lg' (124px) */ size?: 'sm' | 'md' | 'lg'; /** Съдържание — цифра или geometric glyph (никога emoji) */ glyph: number | ReactNode; /** Активно състояние — добавя glow и динамика */ active?: boolean; /** Завършено състояние — потъмнен, dust overlay */ done?: boolean; /** Lock — заключен (за achievements) */ locked?: boolean; /** Accessible label — задължителен ако glyph е number или icon */ ariaLabel: string; }
08 · PhaseStrip · primitives/PhaseStrip.tsx
/** * PhaseStrip — линия от Medallion-и съединени с златна линия. * Употреба: правилата (статично) + in-game (live, signature момент). */ export interface Phase { id: string; label: string; // „Нощ", „Ден", „Глас" sub?: string; // „02:14", „следваща" glyph?: ReactNode; // заместител на цифра } export interface PhaseStripProps { phases: Phase[]; /** Индекс на активната фаза */ activeIndex: number; /** Размер на медальоните */ size?: 'sm' | 'md'; /** Кога една фаза е „в цикъл" (Нощ → Ден → Глас → Нощ) */ loopFrom?: number; } // Активната фаза получава glow + breathing animation (3s, gated by reduced-motion). // Минали фази: opacity 0.45 + dust overlay. // Бъдещи: opacity 0.7, без glow.
09 · PlayerDiary · primitives/PlayerDiary.tsx
/** * PlayerDiary — личният дневник на играч в текущата игра. * Тайното оръжие на социалните игри. След игра се вгражда в делото. */ export interface DiaryEntry { id: string; phase: 'night' | 'day' | 'vote' | 'reveal'; phaseIndex: number; // „Нощ #2", „Ден #1" text: string; // „Иван — невинен" outcome?: 'positive' | 'negative' | 'neutral'; isSecret?: boolean; // само ти го виждаш } export interface PlayerDiaryProps { entries: DiaryEntry[]; /** Дали показваме placeholder за следваща фаза */ showPending?: boolean; /** Compact mode за in-game right rail */ compact?: boolean; }
10 · VoteCircle · primitives/VoteCircle.tsx
/** * VoteCircle — играчите подредени в окръжност (масата). * Tap = vote. Long press = защита (зависи от роля). */ export interface Player { id: string; name: string; avatar?: string; alive: boolean; isYou?: boolean; votesAgainst?: number; } export interface VoteCircleProps { players: Player[]; /** Текуща стъпка — определя кои действия са разрешени */ mode: 'idle' | 'voting' | 'targeting'; /** Кой е избран в момента (един за гласуване, многократно за targeting) */ selectedIds: string[]; onSelect: (playerId: string) => void; /** Може ли да гласува за себе си */ allowSelf?: boolean; } // Кръговото подреждане е CSS трик (rotate + translate). // Под 6 играчи — semicircle отдолу. Над 12 — двоен пръстен.
11 · EmptyState · primitives/EmptyState.tsx
/** * EmptyState — физически артефакт за празно състояние. * Никога generic „No data" иконка. */ export interface EmptyStateProps { /** Кой артефакт — определя SVG илюстрацията */ artifact: | 'sealed-letter' // архив празен | 'empty-chair' // лоби / приятели | 'dusty-shelf' // постижения | 'open-door' // стая празна | 'balanced-scale' // равни резултати | 'unprinted-paper' // leaderboard преди първа игра | 'broken-candle' // сесия изтекла | 'closed-book'; // история празна title: string; description?: string; /** Action бутон, ако empty state има изход */ action?: { label: string; href?: string; onClick?: () => void }; }
12 · Toast · primitives/Toast.tsx
/** * Toast — системно съобщение. Един компонент, три тона. */ export interface ToastProps { /** Тоналност */ tone: 'info' | 'warning' | 'error' | 'success'; /** Заглавие — eyebrow стил */ title: string; /** Тяло — едно изречение, не повече */ description?: string; /** Auto-dismiss след X ms; 0 = persistent */ duration?: number; /** Action бутон отдясно (напр. „Опитай отново") */ action?: { label: string; onClick: () => void }; } // Position: top-center в in-game, bottom-right навсякъде другаде. // Animation: slide-in 200ms cubic-bezier(0.16, 1, 0.3, 1).
13 · Sheet · primitives/Sheet.tsx
/** * Sheet — bottom sheet (mobile) / centered modal (desktop). * Един компонент, две поведения в зависимост от viewport. */ export interface SheetProps { open: boolean; onClose: () => void; /** Заглавие — рендерира се в sheet header */ title?: string; /** Eyebrow над заглавието */ eyebrow?: string; /** Mobile-only: drag-to-dismiss handle */ draggable?: boolean; /** Disable close-on-outside-click (за destructive actions) */ insistent?: boolean; children: ReactNode; } // ESC винаги затваря. // Focus trap е задължителен. // При close focus се връща на trigger.
14 · Dialog · primitives/Dialog.tsx
/** * Dialog — confirm / destructive action. Заразиен потвърждение. * Sheet с предефинирана структура за confirm flow-и. */ export interface DialogProps { open: boolean; onClose: () => void; title: string; description: string; /** Действия — destructive вижда blood; иначе ghost + primary */ confirm: { label: string; onClick: () => void; destructive?: boolean; /** Изисквай typed confirmation („напиши името си") */ typedConfirmation?: string; }; cancel?: { label: string; }; } // destructive=true автоматично прави confirm бутона blood и иска фокус. // typedConfirmation добавя input — submit disabled докато не съвпадне.
03 / Композиция

Как се комбинират.

Три типични страници от продукта — изградени само от primitives. Никакъв друг код освен композиция.

Пример · landing hero (home page)
import { SceneCard, Eyebrow, Display, Pill } from '@village/ui'; export function HomeHero() { return ( <SceneCard density="lg"> <Eyebrow>Избери игра</Eyebrow> <Display size="hero"> Върколак или <em>Мафия</em> </Display> <p>Две отделни игри, два отделни речника, два отделни набора роли.</p> <Pill variant="primary" size="lg" leading="play" href="/play"> Играй </Pill> </SceneCard> ); }
Пример · /play night scene
import { SceneCard, PhaseStrip, Medallion, PlayerDiary, Pill } from '@village/ui'; export function NightScene({ phases, diary, role, onAction }) { return ( <SceneCard density="lg"> <Eyebrow>Нощ #2 · Действай</Eyebrow> <Display size="h1">{role.name}</Display> <p>{role.instruction}</p> <Timer seconds={134} /> <Pill variant="primary" onClick={onAction}>Избери жертва</Pill> <PhaseStrip phases={phases} activeIndex={2} /> <PlayerDiary entries={diary} compact /> </SceneCard> ); }
Пример · /history dело №X карточка
import { PaperCard, Eyebrow, Display, Pill } from '@village/ui'; export function CaseCard({ id, title, faction, date, onOpen }) { return ( <PaperCard eyebrow={`Дело №${id}`} meta={<Eyebrow tone="muted">{date}</Eyebrow>} > <Display size="h3">{title}</Display> <Eyebrow tone="muted">{faction}</Eyebrow> <Pill variant="ghost" size="sm" onClick={onOpen}> Отвори дело → </Pill> </PaperCard> ); }
04 / Дисциплина

Правила за добавяне на нов primitive.

Тези пет правила пазят броя от експлозия.

  • 2x правило. Нов primitive се добавя само ако ще се ползва на 2+ места. На едно място — той е компонент на страницата.
  • Един отговорност. Eyebrow прави text styling. PaperCard прави layout. Не се смесват.
  • Никакви бизнес зависимости. Primitive не знае нищо за gameState, auth, или роли. Само props.
  • Storybook story задължителен. Без story PR се blocking-ва.
  • A11y от старта. Всеки primitive минава axe-core преди да се мерджне.
